        function switchTab(tabName) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.nav-btn').forEach(el => el.classList.remove('active'));
            
            if(tabName === 'reader') {
                document.getElementById('sec-reader').classList.remove('hidden');
            } else {
                document.getElementById('sec-' + tabName).classList.remove('hidden');
                document.getElementById('btn-' + tabName).classList.add('active');
            }
            window.scrollTo(0,0);
            if(tabName === 'admin') {
                if(!isAdminAuthenticated) { switchTab('info'); return; }
                switchAdminSubTab('add-chapter');
                renderAdminManagementLists();
            }
        }

        function switchAdminSubTab(subTabId) {
            document.querySelectorAll('.admin-sub-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('[id^="sub-btn-"]').forEach(el => el.classList.remove('bg-purple-600', 'text-white', 'bg-cyan-500', 'text-black'));
            
            document.getElementById('sub-sec-' + subTabId).classList.remove('hidden');
            if(subTabId === 'manage-items') {
                document.getElementById('sub-btn-' + subTabId).classList.add('bg-purple-600', 'text-white');
            } else {
                document.getElementById('sub-btn-' + subTabId).classList.add('bg-cyan-500', 'text-black');
            }
        }

        function updateAppUI() {

        function handleMangaUpload(e) {
            e.preventDefault();
            const num = parseInt(document.getElementById('ch-number').value);
            const title = document.getElementById('ch-title').value;
            const cover = document.getElementById('ch-cover').value;
            const pagesText = document.getElementById('ch-pages').value;
            const pages = pagesText.split('\n').map(p => p.trim()).filter(p => p.length > 0);

            const mangaData = JSON.parse(localStorage.getItem('imprint_manga'));
            mangaData.push({ num, title: "الفصل " + num + ": " + title, cover, pages });
            mangaData.sort((a,b) => a.num - b.num);

            localStorage.setItem('imprint_manga', JSON.stringify(mangaData));
            alert('تم نشر الفصل الجديد!');
            e.target.reset();
            updateAppUI();
            switchTab('manga');
        }

        function handleNewsUpload(e) {
            e.preventDefault();
            const title = document.getElementById('news-title').value;
            const content = document.getElementById('news-content').value;
            const img = document.getElementById('news-img').value;
            const date = new Date().toISOString().split('T')[0];

            const newsData = JSON.parse(localStorage.getItem('imprint_news'));
            newsData.unshift({ title, content, img, date });

            localStorage.setItem('imprint_news', JSON.stringify(newsData));
            alert('تم بث الخبر!');
            e.target.reset();
            updateAppUI();
            switchTab('news');
        }

        function handleProductUpload(e) {
            e.preventDefault();
            const name = document.getElementById('prod-name').value;
            const price = document.getElementById('prod-price').value;
            const desc = document.getElementById('prod-desc').value;
            const img = document.getElementById('prod-img').value;
            const link = document.getElementById('prod-link').value;

            const storeData = JSON.parse(localStorage.getItem('imprint_store'));
            storeData.push({ name, price, desc, img, link });

            localStorage.setItem('imprint_store', JSON.stringify(storeData));
            alert('تم إدراج المنتج!');
            e.target.reset();
            updateAppUI();
            switchTab('store');
        }

        function deleteItem(type, index) {
            if(confirm('هل أنت متأكد من الحذف؟')) {
                const key = 'imprint_' + type;
                const data = JSON.parse(localStorage.getItem(key));
                data.splice(index, 1);
                localStorage.setItem(key, JSON.stringify(data));
                updateAppUI();
                renderAdminManagementLists();
            }
        }
